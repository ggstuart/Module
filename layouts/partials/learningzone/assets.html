{{- /* Builds (once per build) the extra files a Learning Zone page needs and returns their paths: style the theme's style.css without its Google Fonts @import fonts local copy of those web fonts ("" when offline), fontFiles its files css learningzone.css (iframe-friendly tweaks) js learningzone.js (slide controls, focus, folder links, Mermaid) */ -}} {{- $styleSrc := "" -}} {{- if fileExists "static/css/style.css" -}} {{- $styleSrc = readFile "static/css/style.css" -}} {{- else -}} {{- with resources.Get "_lz/theme-static/css/style.css" }}{{ $styleSrc = .Content }}{{ end -}} {{- end -}} {{- /* The fonts are named in one place: the @import at the top of style.css. */ -}} {{- $import := `@import\s+url\(\s*['"]?(https?://fonts\.googleapis\.com[^)'"]*)['"]?\s*\)\s*;?` -}} {{- $fontsURL := "" -}} {{- with findRESubmatch $import $styleSrc 1 }}{{ $fontsURL = index (index . 0) 1 }}{{ end -}} {{- $styleSrc = replaceRE $import "/* web fonts are bundled by the Learning Zone export */" $styleSrc -}} {{- /* Same folder as the original, so its relative url(../images/...) still work. */ -}} {{- $style := resources.FromString "css/style.lz.css" $styleSrc -}} {{- $fonts := partialCached "learningzone/fonts.html" $fontsURL "lz-fonts" -}} {{- return dict "style" $style.RelPermalink "fonts" $fonts.css "fontFiles" $fonts.files "css" (resources.Get "learningzone/learningzone.css").RelPermalink "js" (resources.Get "learningzone/learningzone.js").RelPermalink -}}
