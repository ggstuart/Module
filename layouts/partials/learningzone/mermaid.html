{{- /* Download Mermaid once at build time and publish it inside the package. A failed download stops the build, unless params.learningzone.offline is true, in which case diagrams are shown as their text source. */ -}} {{- $url := site.Params.learningzone.mermaidURL | default "https://cdn.jsdelivr.net/npm/mermaid@11/dist/mermaid.min.js" -}} {{- $out := "" -}} {{- if not site.Params.learningzone.offline -}} {{- $hint := "Check the internet connection, or build with the learningzone `offline` option (diagrams then show as text)." -}} {{- with try (resources.GetRemote $url) -}} {{- if .Value -}} {{- $out = (resources.Copy "learningzone/js/mermaid.min.js" .Value).RelPermalink -}} {{- else -}} {{- errorf "Learning Zone: could not download Mermaid from %s. %s %s" $url $hint (.Err | default "") -}} {{- end -}} {{- end -}} {{- end -}} {{- return $out -}}
